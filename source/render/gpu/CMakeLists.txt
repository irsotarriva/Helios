# The bgfx renderer of the map view. Only built with HELIOS_BUILD_CLIENT.
include(${bgfx_SOURCE_DIR}/cmake/bgfxToolUtils.cmake)
set(BGFX_SHADER_INCLUDE_PATH ${bgfx_SOURCE_DIR}/bgfx/src)

# Shader binaries for the back-ends each platform uses (Metal on macOS; D3D, Vulkan and OpenGL on
# Windows; Vulkan and OpenGL on Linux), compiled by shaderc at build time and embedded as headers.
if(APPLE)
  set(helios_shader_profiles metal)
elseif(WIN32)
  set(helios_shader_profiles s_5_0 s_6_0 spirv 430)
else()
  set(helios_shader_profiles spirv 430)
endif()

set(helios_shader_dir ${CMAKE_CURRENT_BINARY_DIR}/shaders)
set(helios_shader_source ${CMAKE_CURRENT_SOURCE_DIR}/shaders)
bgfx_compile_shaders(TYPE VERTEX SHADERS ${helios_shader_source}/vs_body.sc ${helios_shader_source}/vs_line.sc
  ${helios_shader_source}/vs_solid.sc
  VARYING_DEF ${helios_shader_source}/varying.def.sc OUTPUT_DIR ${helios_shader_dir}
  PROFILES ${helios_shader_profiles} AS_HEADERS OUT_FILES_VAR helios_vs_scene)
bgfx_compile_shaders(TYPE FRAGMENT SHADERS ${helios_shader_source}/fs_body.sc ${helios_shader_source}/fs_line.sc
  ${helios_shader_source}/fs_solid.sc
  VARYING_DEF ${helios_shader_source}/varying.def.sc OUTPUT_DIR ${helios_shader_dir}
  PROFILES ${helios_shader_profiles} AS_HEADERS OUT_FILES_VAR helios_fs_scene)
bgfx_compile_shaders(TYPE VERTEX SHADERS ${helios_shader_source}/vs_imgui.sc
  VARYING_DEF ${helios_shader_source}/varying_imgui.def.sc OUTPUT_DIR ${helios_shader_dir}
  PROFILES ${helios_shader_profiles} AS_HEADERS OUT_FILES_VAR helios_vs_imgui)
bgfx_compile_shaders(TYPE FRAGMENT SHADERS ${helios_shader_source}/fs_imgui.sc
  VARYING_DEF ${helios_shader_source}/varying_imgui.def.sc OUTPUT_DIR ${helios_shader_dir}
  PROFILES ${helios_shader_profiles} AS_HEADERS OUT_FILES_VAR helios_fs_imgui)

# One generated table of every (shader, back-end) blob, so C++ never names a profile.
set(helios_shader_names vs_body fs_body vs_line fs_line vs_solid fs_solid vs_imgui fs_imgui)
set(table_includes "")
set(table_rows "")
foreach(profile ${helios_shader_profiles})
  if(profile STREQUAL "430")
    set(directory glsl)
    set(suffix glsl)
    set(renderer OpenGL)
  elseif(profile STREQUAL "spirv")
    set(directory spirv)
    set(suffix spv)
    set(renderer Vulkan)
  elseif(profile STREQUAL "metal")
    set(directory metal)
    set(suffix mtl)
    set(renderer Metal)
  elseif(profile STREQUAL "s_5_0")
    set(directory dxbc)
    set(suffix dxbc)
    set(renderer Direct3D11)
  elseif(profile STREQUAL "s_6_0")
    set(directory dxil)
    set(suffix dxil)
    set(renderer Direct3D12)
  endif()
  foreach(name ${helios_shader_names})
    string(APPEND table_includes "#include \"${directory}/${name}.sc.bin.h\"\n")
    string(APPEND table_rows "    {\"${name}\", bgfx::RendererType::${renderer}, ${name}_${suffix}, sizeof(${name}_${suffix})},\n")
  endforeach()
endforeach()
file(GENERATE OUTPUT ${helios_shader_dir}/embedded_shaders.inc CONTENT
"// Generated by source/render/gpu/CMakeLists.txt. Do not edit.
${table_includes}
namespace helios::render::gpu {
inline constexpr EmbeddedShader k_embedded_shaders[] = {
${table_rows}};
} // namespace helios::render::gpu
")

add_library(helios_render_gpu
  embedded_shaders.cpp
  eye_target.cpp
  imgui_renderer.cpp
  map_renderer.cpp
  solid_renderer.cpp
  ${helios_vs_scene} ${helios_fs_scene} ${helios_vs_imgui} ${helios_fs_imgui}
)
add_library(helios::render_gpu ALIAS helios_render_gpu)
target_include_directories(helios_render_gpu
  PUBLIC  ${PROJECT_SOURCE_DIR}/include
  PRIVATE ${helios_shader_dir}
)
target_link_libraries(helios_render_gpu
  PUBLIC  helios::render_core bgfx bx bimg helios::imgui
  PRIVATE helios::compile_options
)
